    message(FATAL_ERROR "This port requires Apple Silicon macOS")

execute_process(COMMAND brew --prefix OUTPUT_VARIABLE BREW_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)

include(FetchContent)

set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sdl2 URL https://github.com/libsdl-org/SDL/archive/refs/tags/release-2.30.12.tar.gz URL_HASH SHA256=560da2e54dd8af933e35bd08fb1b6cf80d4f6938c67710fecf13b7e9bdd6c47e DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(sdl2)
set(MI_OVERRIDE OFF CACHE BOOL "" FORCE)
set(MI_OSX_INTERPOSE OFF CACHE BOOL "" FORCE)
set(MI_OSX_ZONE OFF CACHE BOOL "" FORCE)
set(MI_BUILD_SHARED OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(spdlog_source)

include(ExternalProject)
ExternalProject_Add(dxvk-native
    SOURCE_DIR "${CMAKE_SOURCE_DIR}/dxvk"
    BINARY_DIR "${CMAKE_BINARY_DIR}/dxvk"
    DOWNLOAD_COMMAND ""
    UPDATE_COMMAND ""
    CONFIGURE_COMMAND "${CMAKE_COMMAND}" -E env
        "CFLAGS=-I${CMAKE_BINARY_DIR}/include -I${BREW_PREFIX}/include"
        "CXXFLAGS=-I${CMAKE_BINARY_DIR}/include -I${BREW_PREFIX}/include"
        "LDFLAGS=-L${CMAKE_BINARY_DIR}/bin -L${BREW_PREFIX}/lib -Wl,-rpath,${CMAKE_BINARY_DIR}/bin"
        "PKG_CONFIG_PATH=${BREW_PREFIX}/lib/pkgconfig"
        meson setup <BINARY_DIR> <SOURCE_DIR> --buildtype=release
        -Denable_d3d11=false -Denable_d3d10=false -Denable_dxgi=false
        -Denable_tests=false -Dnative_sdl2_library=$<TARGET_FILE:SDL2>
    BUILD_COMMAND ninja -C <BINARY_DIR> -j2
    BUILD_BYPRODUCTS "${CMAKE_BINARY_DIR}/dxvk/src/d3d9/libdxvk_d3d9.dylib"
    INSTALL_COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${CMAKE_BINARY_DIR}/dxvk/src/d3d9/libdxvk_d3d9.dylib" "${CMAKE_BINARY_DIR}/bin/"
    DEPENDS SDL2)
set(NATIVE_D3D9_LIBS "${CMAKE_BINARY_DIR}/dxvk/src/d3d9/libdxvk_d3d9.dylib")
include_directories("${CMAKE_SOURCE_DIR}/dxvk/include/native/directx" "${CMAKE_SOURCE_DIR}/dxvk/include/native/windows" "${CMAKE_SOURCE_DIR}/dxvk/src")
